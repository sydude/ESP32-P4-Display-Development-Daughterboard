cmake_minimum_required(VERSION 3.16)
project(brightness_host_tests C CXX)
set(CMAKE_C_STANDARD 11)
get_filename_component(APP "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP}/managed_components/lvgl__lvgl" lvgl)
file(GLOB_RECURSE UI_SOURCES "${APP}/main/ui/*.c")
add_executable(brightness_test brightness_test.c "${APP}/main/brightness_control.c" ${UI_SOURCES})
target_include_directories(brightness_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${APP}/main" "${APP}/main/ui")
target_compile_definitions(brightness_test PRIVATE LV_LVGL_H_INCLUDE_SIMPLE=1)
target_link_libraries(brightness_test PRIVATE lvgl)
enable_testing()
foreach(mode success probe_failure init_write_failure update_failure)
    add_test(NAME ${mode} COMMAND brightness_test ${mode})
endforeach()

add_executable(diagnostic_test brightness_test.c "${APP}/main/brightness_control.c" ${UI_SOURCES})
target_include_directories(diagnostic_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${APP}/main" "${APP}/main/ui")
target_compile_definitions(diagnostic_test PRIVATE LV_LVGL_H_INCLUDE_SIMPLE=1 CONFIG_BACKLIGHT_DIAGNOSTIC=1)
target_link_libraries(diagnostic_test PRIVATE lvgl)
foreach(mode diagnostic_success diagnostic_failure diagnostic_mid_failure probe_failure task_failure)
    add_test(NAME diag_${mode} COMMAND diagnostic_test ${mode})
endforeach()
